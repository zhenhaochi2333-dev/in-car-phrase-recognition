cmake_minimum_required(VERSION 3.20)
project(phrase_lab_core LANGUAGES CXX)
add_library(phrase_lab_core SHARED core.cpp)
target_compile_features(phrase_lab_core PRIVATE cxx_std_17)
target_include_directories(phrase_lab_core PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
set_target_properties(phrase_lab_core PROPERTIES PREFIX "")
if(MSVC)
  target_compile_options(phrase_lab_core PRIVATE /W4 /O2 /utf-8)
else()
  target_compile_options(phrase_lab_core PRIVATE -Wall -Wextra -Wpedantic -O2)
  if(MINGW)
    target_link_options(phrase_lab_core PRIVATE -static-libgcc -static-libstdc++ -static)
  endif()
endif()
